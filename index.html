function startSupportChat(){ const uid = document.getElementById("supportUID")?.value.trim(); const problem = document.getElementById("supportProblem")?.value; if(!uid){ alert("🆔 আগে আপনার UID দিন।"); return; } if(!problem){ alert("📋 আগে সমস্যাটি Select করুন।"); return; } let supportUserId = localStorage.getItem("taskmoon_support_user_id"); if(!supportUserId){ supportUserId = String(Date.now()); localStorage.setItem( "taskmoon_support_user_id", supportUserId ); } fetch( "https://taskmoon-backend.onrender.com/support/chat", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ user_id: Number(supportUserId), uid: uid, problem: problem, message: "" }) } ) .then(async response => { const data = await response.json(); if(!response.ok || !data.success){ throw new Error( data.error || "Chat শুরু করা যায়নি" ); } return data; }) .then(data => { const box = document.getElementById("supportChatBox"); if(!box) return; box.innerHTML = `

💬 TaskMoon Live Chat

🆔 UID: ${escapeHtml(uid)}

📋 Problem: ${escapeHtml(problem)}

🟢 Chat Started!
Admin-এর reply-এর জন্য অপেক্ষা করুন।
`; startSupportChatRefresh(data.chat_id); }) .catch(error => { console.error( "Support chat error:", error ); alert( "❌ Server-এর সাথে যোগাযোগ করা যাচ্ছে না।\n" + error.message ); }); }
